rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {


    // Helpers

    function isSignedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return isSignedIn()
        && request.auth.token.email == "canteenadmin1@gmail.com";
    }

    function isSelf(userId) {
      return isSignedIn() && request.auth.uid == userId;
    }


    // USERS  (single collection, role: 'user' | 'admin')

    match /users/{userId} {

      allow read: if true;

      // ── Create ──
      allow create: if isSelf(userId) || isAdmin();
      allow update: if isSelf(userId) || isAdmin();
      allow delete: if isAdmin();
    }


    // CANTEEN

    match /canteen/{document} {
      allow read: if true;
      allow write: if isAdmin();
    }


    // MENU ITEMS

    match /menu_items/{document} {
      allow read: if isSignedIn();
      allow write: if isAdmin();

      // Students can decrement stock / preparedPortions when ordering.
      allow update: if isSignedIn()
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['stock', 'isAvailable'])
        && request.resource.data.stock is int
        && request.resource.data.stock < resource.data.stock
        && request.resource.data.stock >= 0;

      allow update: if isSignedIn()
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['preparedPortions', 'isAvailable'])
        && request.resource.data.preparedPortions is int
        && request.resource.data.preparedPortions < resource.data.preparedPortions
        && request.resource.data.preparedPortions >= 0;
    }


    // INVENTORY

    match /inventory/{itemId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }


    // ORDERS

    match /orders/{orderId} {
      allow read: if isSignedIn()
        && (request.auth.uid == resource.data.userId || isAdmin());

      allow create: if isSignedIn()
        && request.resource.data.userId == request.auth.uid;

      allow update: if isAdmin()
        || (isSignedIn()
            && request.auth.uid == resource.data.userId
            && resource.data.status == 'Pending'
            && request.resource.data.status == 'Cancelled');

      allow delete: if isAdmin();
    }


    // CREDIT TRANSACTIONS

    match /credit_transactions/{txId} {
      allow read: if isSignedIn()
        && (request.auth.uid == resource.data.uid || isAdmin());

      allow create: if isSignedIn()
        && (request.resource.data.uid == request.auth.uid || isAdmin());

      allow update, delete: if false;
    }


    // INVENTORY UNITS

    match /inventory_units/{unitId} {
      allow read: if isSignedIn();
      allow create: if isAdmin();
      allow update, delete: if false;
    }


    // LOYALTY REWARDS

    match /loyalty_rewards/{rewardId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }


    // POINTS TRANSACTIONS

    match /points_transactions/{txId} {
      allow read: if isSignedIn()
        && (request.auth.uid == resource.data.uid || isAdmin());

      allow create: if isSignedIn()
        && (request.resource.data.uid == request.auth.uid || isAdmin());

      allow update, delete: if false;
    }


    // REDEMPTIONS

    match /redemptions/{redemptionId} {
      allow read: if isSignedIn()
        && (request.auth.uid == resource.data.userId || isAdmin());

      allow create: if isSignedIn()
        && request.resource.data.userId == request.auth.uid;

      allow update: if isAdmin();

      allow delete: if false;
    }
  }
}
